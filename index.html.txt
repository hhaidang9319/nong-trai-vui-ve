        function checkCurrentEvent() {
            const now = new Date();
            const month = now.getMonth() + 1;
            const date = now.getDate();
            const dayOfWeek = now.getDay();
            const isWeekend = (dayOfWeek === 0 || dayOfWeek === 6);

            let eventName = "Ngày Thường Vui Vẻ";
            let desc = "Chăm chỉ cày cuốc để thăng cấp nông trại!";
            let rewardItemKey = "seed_tet_mai";

            if (month === 1 && date <= 10) { eventName = "Tết Nguyên Đán 🌸"; desc = "Nhận lì xì vàng & hoa mai vàng nở rộ!"; rewardItemKey = "seed_tet_mai"; }
            else if (month === 3 && date === 10) { eventName = "Giỗ Tổ Hùng Vương 🇻🇳"; desc = "Đất nước hướng về cội nguồn, tăng EXP thu hoạch!"; rewardItemKey = "seed_hungvuong_sen"; }
            else if (month === 4 && (date === 30 || date === 1)) { eventName = "Lễ 30/4 - 1/5 🚩"; desc = "Đại lễ thống nhất đất nước, x2 năng suất lao động!"; rewardItemKey = "seed_hungvuong_sen"; }
            else if (month === 9 && date === 2) { eventName = "Quốc Khánh 2/9 ⭐"; desc = "Mừng ngày độc lập, quà tặng ngập tràn!"; rewardItemKey = "seed_hungvuong_sen"; }
            else if (month === 9 && date >= 10 && date <= 17) { eventName = "Tết Trung Thu 🥮"; desc = "Lồng đèn thắp sáng, ghép cá nhận quà trăng rằm!"; rewardItemKey = "seed_trungthu_moon"; }
            else if (month === 11 && date === 20) { eventName = "Ngày Nhà Giáo Việt Nam 20/11 💐"; desc = "Tặng hoa tri ân, nhận thưởng đặc biệt!"; rewardItemKey = "seed_tet_mai"; }
            else if (isWeekend) { eventName = "Sự Kiện Cuối Tuần (Thứ 7 & Chủ Nhật) 🎉"; desc = "Kích hoạt X2 Vàng, Giảm 30% thời gian & Buff +20% tỷ lệ ép cá!"; rewardItemKey = "fish_hoalinh_carp"; }

            return { isWeekend, eventName, desc, rewardItemKey };
        }

        function renderFarm() {
            const grid = document.getElementById('farm-grid');
            grid.innerHTML = '';
            let unlockedCount = 0;
            gameState.farmPlots.forEach((plot, index) => {
                if(plot.state !== 'locked') unlockedCount++;
                let content = '';
                let bgClass = 'bg-amber-200 border-amber-400';
                
                if (plot.state === 'locked') {
                    bgClass = 'bg-gray-200 border-gray-300 opacity-60';
                    content = `<div class="text-center">🔒<p class="text-xs mt-1 font-semibold">Khóa</p></div>`;
                } else if (plot.state === 'empty') {
                    // Kiểm tra xem đang chọn hạt giống nào (thường hoặc độc quyền lễ)
                    const activeSeed = seedCatalog[gameState.selectedSeed] || exclusiveHolidayItems[gameState.selectedSeed];

                } else if (plot.state === 'growing') {
                    const elapsed = Math.floor((Date.now() - plot.plantedTime) / 1000);
                    const left = Math.max(0, plot.duration - elapsed);
                    const item = seedCatalog[plot.seedKey] || exclusiveHolidayItems[plot.seedKey];
                    content = `<div class="text-center cursor-pointer" onclick="speedUpFarm(${index})">${item ? item.icon : '🌱'}<p class="text-xs text-amber-900 font-semibold mt-1">Đang lớn (${left}s)</p><p class="text-[9px] text-cyan-600 font-bold underline">⚡ Tăng tốc (10💎)</p></div>`;
                } else if (plot.state === 'ready') {
                    bgClass = 'bg-emerald-100 border-emerald-400 animate-pulse';
                    const item = seedCatalog[plot.seedKey] || exclusiveHolidayItems[plot.seedKey];
                    content = `<div class="text-center cursor-pointer">✨${item ? item.icon : '🥬'}<p class="text-xs text-emerald-800 font-bold mt-1">Thu hoạch!</p></div>`;
                }

                const card = document.createElement('div');
                card.className = `h-36 rounded-2xl border-2 ${bgClass} flex flex-col items-center justify-center shadow-sm transition hover:scale-[1.02]`;
                card.innerHTML = content;
                card.onclick = () => { if(plot.state === 'growing') return; handleFarmClick(index); };
                grid.appendChild(card);
            });

        function handleFarmClick(index) {
            const plot = gameState.farmPlots[index];
            if (plot.state === 'locked') {
                showToast("Ô đất này chưa mở khóa!");
            } else if (plot.state === 'empty') {
                const seedKey = gameState.selectedSeed;
                if ((gameState.inventory[seedKey] || 0) > 0) {
                    gameState.inventory[seedKey]--;
                    const item = seedCatalog[seedKey] || exclusiveHolidayItems[seedKey];
                    plot.state = 'growing';
                    plot.seedKey = seedKey;
                    plot.plantedTime = Date.now();
                    
                    const ev = checkCurrentEvent();
                    let finalDuration = item.duration;
                    if(ev.isWeekend) finalDuration = Math.floor(item.duration * 0.7);

                    plot.duration = finalDuration;
                    showToast(`Đã gieo trồng ${item.name}!`);
                    renderFarm();
                } else {
                    showToast("Bạn đã hết hạt giống này trong kho!");
                }
            } else if (plot.state === 'ready') {
                if (getTotalInventoryCount() >= gameState.inventoryCapacity) {
                    showToast("🎒 Kho chứa đã đầy!");
                    return;
                }
                const item = seedCatalog[plot.seedKey] || exclusiveHolidayItems[plot.seedKey];
                plot.state = 'empty';
                
                const ev = checkCurrentEvent();
                let baseReward = item.reward || 100;
                if(ev.isWeekend) baseReward *= 2;

                // Thưởng vàng tăng theo cấp độ lâu đài (mỗi level lâu đài tăng 2% giá trị)
                let finalGold = Math.floor(baseReward * (1 + gameState.villaLevel * 0.02));

                gameState.gold += finalGold;
                gameState.exp += (item.exp || 30);
                const harvestKey = 'harvest_' + plot.seedKey.replace('seed_', '').replace('fish_', '');
                gameState.inventory[harvestKey] = (gameState.inventory[harvestKey] || 0) + 1;
                checkLevelUp();
                showToast(`Thu hoạch thành công +${finalGold} Vàng!`);
                updateHeader();
                renderFarm();
            }
        }

        function fuseFish(recipe, useDiamond) {
            const ev = checkCurrentEvent();
            if (recipe === 'recipe1') {
                if ((gameState.inventory['harvest_carp'] || 0) >= 2) {
                    if (getTotalInventoryCount() >= gameState.inventoryCapacity) { showToast("Kho đã đầy!"); return; }
                    gameState.inventory['harvest_carp'] -= 2;
                    
                    let rate = 0.6;
                    if(ev.isWeekend) rate += 0.2;
                    let success = useDiamond ? true : (Math.random() < rate);

                    if(useDiamond) {
                        if(gameState.diamond >= 20) gameState.diamond -= 20;
                        else { showToast("Không đủ 20 Kim Cương!"); gameState.inventory['harvest_carp'] += 2; return; }
                    }
                    if (success) {
                        gameState.inventory['rare_dragon_fish'] = (gameState.inventory['rare_dragon_fish'] || 0) + 1;
                        showToast("🎉 Ép thành công Cá Rồng Hoàng Kim!");
                    } else { showToast("😢 Ép cá thất bại!"); }
                    updateHeader();
                    renderInventory();
                } else { showToast("Cần ít nhất 2 Cá Chép Trưởng Thành!"); }
            } else if (recipe === 'recipe2') {
                if ((gameState.inventory['harvest_salmon'] || 0) >= 1 && (gameState.inventory['harvest_puffer'] || 0) >= 1) {
                    if (getTotalInventoryCount() >= gameState.inventoryCapacity) { showToast("Kho đã đầy!"); return; }
                    gameState.inventory['harvest_salmon']--;
                    gameState.inventory['harvest_puffer']--;
                    
                    let rate = 0.5;
                    if(ev.isWeekend) rate += 0.2;
                    let success = useDiamond ? true : (Math.random() < rate);

                    if(useDiamond) {
                        if(gameState.diamond >= 30) gameState.diamond -= 30;
                        else { showToast("Không đủ 30 Kim Cương!"); gameState.inventory['harvest_salmon']++; gameState.inventory['harvest_puffer']++; return; }
                    }
                    if (success) {
                        gameState.inventory['rare_rainbow_fish'] = (gameState.inventory['rare_rainbow_fish'] || 0) + 1;
                        showToast("🎉 Ép thành công Cá Thần Bảy Sắc!");
                    } else { showToast("😢 Ép cá thất bại!"); }
                    updateHeader();
                    renderInventory();
                } else { showToast("Cần 1 Cá Hồi + 1 Cá Nóc Trưởng Thành!"); }
            }
        }

        }

                gameState.diamond -= 100;

                updateHeader();
                showToast("🐕 Đã nâng cấp Thần Khuyển Địa Ngục!");

        }

        function renderLogs() {
            const logBox = document.getElementById('activity-log');
            if(logBox) logBox.innerHTML = gameState.logs.map(l => `<p>• ${l}</p>`).join('');
        }

        // Lâu Đài Tối Đa Lv.30
        function renderVilla() {
            const currentLevel = gameState.villaLevel;
            let icon = '🏡'; let title = 'Nhà Tranh Cơ Bản';
            if(currentLevel >= 10 && currentLevel < 20) { icon = '🏰'; title = 'Lâu Đài Vàng Rồng'; }
            else if(currentLevel >= 20) { icon = '👑'; title = 'Cung Điện Thiên Giới'; }

            document.getElementById('villa-icon').innerText = icon;
            document.getElementById('villa-title').innerText = `${title} (Lv.${currentLevel}/30)`;
            document.getElementById('villa-desc').innerText = `Đang tăng +${currentLevel * 2}% giá trị thu hoạch nông sản & phong thủy tuyệt đối.`;

            if(currentLevel < 30) {
                const nextGoldCost = (currentLevel + 1) * 1000;
                const nextDiamondCost = (currentLevel + 1) * 20;
                document.getElementById('villa-cost').innerText = `${nextGoldCost} Vàng / ${nextDiamondCost} 💎`;
            } else {
                document.getElementById('villa-cost').innerText = `Đã đạt cấp độ tối đa (Lv.30)!`;
            }
        }

        function upgradeVilla(type) {
            if(gameState.villaLevel >= 30) { showToast("Lâu Đài đã đạt cấp tối đa Lv.30!"); return; }
            const nextLevel = gameState.villaLevel + 1;
            const goldCost = nextLevel * 1000;
            const diamondCost = nextLevel * 20;

            if(type === 'gold') {
                if(gameState.gold >= goldCost) {
                    gameState.gold -= goldCost;
                    gameState.villaLevel++;
                    updateHeader();
                    renderVilla();
                    showToast(`🏰 Nâng cấp Lâu Đài thành công lên Lv.${gameState.villaLevel}!`);
                } else { showToast(`Cần ${goldCost} Vàng!`); }
            } else if(type === 'diamond') {
                if(gameState.diamond >= diamondCost) {
                    gameState.diamond -= diamondCost;
                    gameState.villaLevel++;
                    updateHeader();
                    renderVilla();
                    showToast(`🏰 Nâng cấp Lâu Đài thành công lên Lv.${gameState.villaLevel}!`);
                } else { showToast(`Cần ${diamondCost} Kim Cương!`); }
            }
        }

        function renderCollection() {

        function renderShop() {
            const list = document.getElementById('shop-list');
            list.innerHTML = '';
            for (const [key, item] of Object.entries(seedCatalog)) {
                const isLocked = gameState.level < item.level;
                const card = document.createElement('div');
                card.className = `bg-white p-4 rounded-2xl border ${isLocked ? 'border-gray-300 opacity-60' : 'border-emerald-200'} shadow-sm flex items-center justify-between`;
                card.innerHTML = `
                    <div class="flex items-center space-x-3">
                        <span class="text-3xl">${item.icon}</span>
                        <div>
                            <h4 class="font-bold text-slate-800 text-sm">${item.name}</h4>
                            <p class="text-xs text-emerald-600 font-semibold">Giá: ${item.price} Vàng</p>
                            ${isLocked ? `<p class="text-[10px] text-red-500 font-bold">Yêu cầu Level ${item.level}</p>` : `<p class="text-[10px] text-gray-400">Thu: +${item.reward} V</p>`}
                        </div>
                    </div>
                    <button onclick="buyItem('${key}', ${item.price}, ${item.level})" class="${isLocked ? 'bg-gray-400 cursor-not-allowed' : 'bg-emerald-600 hover:bg-emerald-700'} text-white px-4 py-2 rounded-xl text-xs font-bold shadow">
                        ${isLocked ? 'Khóa' : 'Mua'}
                    </button>
                `;
                list.appendChild(card);
            }
        }

        function buyItem(itemId, price, reqLevel) {
            if (gameState.level < reqLevel) { showToast(`Cần đạt Level ${reqLevel}!`); return; }
            if (getTotalInventoryCount() >= gameState.inventoryCapacity) { showToast("Kho đã đầy!"); return; }
            if (gameState.gold >= price) {
                gameState.gold -= price;
                gameState.inventory[itemId] = (gameState.inventory[itemId] || 0) + 1;
                gameState.selectedSeed = itemId;
                showToast(`Đã mua 1x ${seedCatalog[itemId].name}!`);
                updateHeader();
                renderShop();
            } else { showToast("Không đủ Vàng!"); }
        }

        function renderInventory() {
            const grid = document.getElementById('inventory-grid');
            grid.innerHTML = '';
            document.getElementById('inventory-capacity').innerText = gameState.inventoryCapacity;

            const allItems = {...seedCatalog, ...exclusiveHolidayItems};

            for (const [key, qty] of Object.entries(gameState.inventory)) {
                if (qty > 0) {
                    let displayName = key;
                    let icon = '📦';
                    if(allItems[key]) { displayName = allItems[key].name; icon = allItems[key].icon; }
                    else if(key.startsWith('harvest_')) { displayName = 'Thu: ' + key.replace('harvest_', '').toUpperCase(); icon = '✨'; }
                    else if(key === 'rare_dragon_fish') { displayName = 'Cá Rồng Hoàng Kim'; icon = '🐉'; }
                    else if(key === 'rare_rainbow_fish') { displayName = 'Cá Thần Bảy Sắc'; icon = '🌈'; }

                    const card = document.createElement('div');
                    card.className = 'bg-white p-4 rounded-2xl border border-gray-200 shadow-sm text-center flex flex-col justify-between';
                    card.innerHTML = `
                        <div class="text-3xl mb-1">${icon}</div>
                        <h4 class="font-bold text-xs text-slate-800">${displayName}</h4>
                        <p class="text-xs text-emerald-600 font-bold mt-2">Số lượng: ${qty}</p>
                        <button onclick="selectSeedForPlanting('${key}')" class="mt-2 bg-amber-500 hover:bg-amber-600 text-white text-[10px] py-1 px-2 rounded font-bold">Chọn Trồng</button>
                    `;
                    grid.appendChild(card);
                }
            }
        }

        function selectSeedForPlanting(seedKey) {
            gameState.selectedSeed = seedKey;
            showToast(`Đã chọn ${seedCatalog[seedKey]?.name || exclusiveHolidayItems[seedKey]?.name || seedKey} để gieo trồng!`);
            switchTab('farm');
        }

        function checkLevelUp() {
            const reqExp = gameState.level * 150;
            if (gameState.exp >= reqExp && gameState.level < 100) {
                gameState.level++;
                gameState.exp -= reqExp;
                showToast(`🎉 Chúc mừng bệ hạ thăng cấp ${gameState.level}/100!`);
                updateHeader();
            }
        }

        setInterval(() => {
            let needUpdate = false;
            const now = Date.now();
            gameState.farmPlots.forEach(plot => {
                if (plot.state === 'growing' && Math.floor((now - plot.plantedTime) / 1000) >= plot.duration) {
                    plot.state = 'ready'; needUpdate = true;
                }
            });
            gameState.pondSlots.forEach(slot => {
                if (slot.state === 'growing' && Math.floor((now - slot.fedTime) / 1000) >= slot.duration) {
                    slot.state = 'ready'; needUpdate = true;
                }
            });
            if (needUpdate) {
                if (!document.getElementById('tab-farm').classList.contains('hidden')) renderFarm();
                if (!document.getElementById('tab-pond').classList.contains('hidden')) renderPond();
            }
        }, 1000);

        updateHeader();
        renderFarm();
    </script>
